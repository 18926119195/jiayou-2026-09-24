## 提取算法

### 算法 1：双括号 chunk 抽取（L24）

```python
import re

def extract_l24_chunks(l24_text):
    """提取 L24 中所有路径-步骤-chunk 的引用"""
    pattern = r"#\s*路径([一二三四五六七八九十]+)：([^#]+)"
    path_pat = re.compile(pattern)

    paths = []
    for path_match in path_pat.finditer(l24_text):
        path_id = path_match.group(1)        # "一" "二"...
        path_title = path_match.group(2).split('\n')[0].strip()

        # 提取路径内的步骤块
        path_block = path_match.group(2)
        step_pat = re.compile(r"步骤(\d+)：([^步骤]+)")
        steps = []
        for step_match in step_pat.finditer(path_block):
            step_id = step_match.group(1)
            step_text = step_match.group(2)

            # 提取步骤中的 chunks (双括号)
            chunks = re.findall(r"【【(\d+)】】", step_text)
            steps.append({
                "step_id": step_id,
                "chunks": chunks,
                "text": step_text
            })

        paths.append({
            "path_id": path_id,
            "title": path_title,
            "steps": steps
        })
    return paths
```
